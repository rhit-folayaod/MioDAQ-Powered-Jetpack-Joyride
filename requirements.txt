# Pinned to the versions this was developed and tested against
# (Python 3.14.6 on Windows 11).
#
#     pip install -r requirements.txt

# The game itself. pygame-ce rather than pygame: on Python 3.14 there is no
# prebuilt pygame wheel for cp314, so pip falls back to a source build that
# needs MSYS2. pygame-ce ships a cp314 wheel and is a drop-in `import pygame`.
pygame-ce==2.5.7

# NI DAQ support. Optional at runtime -- the game catches both a missing import
# and a missing device, and falls back to keyboard-only mode either way. Safe to
# install on a machine with no NI hardware or drivers; it simply finds no device.
nidaqmx==1.5.0
